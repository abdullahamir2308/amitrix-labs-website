## Give the agent tools, not freedom

[Placeholder] Agents become useful when they can act — and safe when every action is scoped. A typed tool definition makes the contract explicit:

```ts title="tools/lookup-order.ts"
import { z } from "zod";

export const lookupOrder = {
  name: "lookup_order",
  description: "Fetch an order by id. Read-only.",
  input: z.object({ orderId: z.string().min(1) }),
  async run({ orderId }: { orderId: string }) {
    const order = await db.orders.findUnique({ where: { id: orderId } });
    if (!order) return { found: false } as const;
    return { found: true, status: order.status, eta: order.eta } as const;
  },
};
```
